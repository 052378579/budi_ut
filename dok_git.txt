# 1. Masuk ke direktori proyek
cd /var/www/budi_ut

# 2. (Opsional) Cek berkas apa saja yang telah dimodifikasi
git status

# 3. Masukkan seluruh perubahan ke area staging
git add .

# 4. Buat commit dengan pesan deskriptif (sesuaikan tanggal / nama fitur)
git commit -m "Update: Pembaruan fitur pendaftaran dan validasi"

# 5. Dorong perubahan ke cabang 'dev' di GitHub
git push origin dev

# 6. (Opsional) Jika ingin cabang 'main' langsung sinkron sama persis dengan 'dev'
git push origin dev:main
